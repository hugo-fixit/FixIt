{{- /* Post disclaimer/declaration partial. Displays a lightweight disclaimer notice at the beginning of an article. Supports multiple types: ai, ai-assisted, ai-translated, repost, original, custom. The "custom" type requires content to be specified, otherwise it will not be displayed. The disclaimer content is rendered as Markdown. */ -}} {{- $disclaimer := dict "Page" . "Key" "disclaimer" | partial "function/param.html" | default dict -}} {{- if not $disclaimer.enable -}} {{- return -}} {{- end -}} {{- $type := $disclaimer.type | default "ai" -}} {{- $content := "" -}} {{- if eq $type "custom" -}} {{- /* custom type requires content, otherwise skip */ -}} {{- $content = $disclaimer.content | default "" -}} {{- else -}} {{- $content = $disclaimer.content | default (T (printf "single.disclaimerMessage.%s" $type)) -}} {{- end -}} {{- if not $content -}} {{- return -}} {{- end -}} {{- $types := slice "ai" "repost" "original" -}} {{- $iconSrc := "" -}} {{- if in $types $type -}} {{- $iconSrc = printf "images/icons/disclaimer/%v.svg" $type -}} {{- end -}} {{- $iconOptions := dict "Class" "icon fa-solid fa-bullhorn" -}} {{- if $iconSrc -}} {{- $iconOptions = dict "Src" $iconSrc -}} {{- end -}} {{- $icon := partial "plugin/icon.html" $iconOptions -}} {{- $text := printf `%s%s%s` $icon (T "single.disclaimer") ($content | .RenderString) -}} {{- $admonition := dict "Type" "quote" "Title" "~" "Foldable" false "Class" (printf "post-disclaimer post-disclaimer__%s" $type) "Text" $text -}} {{- partial "plugin/admonition.html" $admonition -}}
