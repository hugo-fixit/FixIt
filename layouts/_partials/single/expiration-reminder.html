{{- /* Content expiration reminder partial. Displays a note or warning admonition when the page's last modification date exceeds configurable day thresholds (default: 90 days for note, 180 days for warning). Skipped for password-protected pages. */ -}} {{- $expirationReminder := dict "Page" . "Key" "expiration_reminder" | partial "function/param.html" | default dict -}} {{- if not ($expirationReminder.enable | and (not .Params.password)) -}} {{- return -}} {{- end -}} {{- $daysAgo := div (sub now.Unix .Lastmod.Unix) 86400 }} {{- $reminderThreshold := $expirationReminder.reminder | default 90 }} {{- $warningThreshold := $expirationReminder.warning | default 180 }} {{- $updateTime := .Lastmod }} {{- if gt $daysAgo $reminderThreshold -}} {{- $type := "note" -}} {{- if gt $daysAgo $warningThreshold -}} {{- $type = "warning" -}} {{- end -}} {{- with $updateTime | dateFormat (.Site.Params.date_format | default "2006-01-02") -}} {{- $message := dict "Date" . | T "single.expirationReminder" -}} {{- dict "Type" $type "Title" "~" "Foldable" false "Class" "expiration-reminder" "Text" $message | partial "plugin/admonition.html" -}} {{- end -}} {{- end -}}
